cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" IMAGO_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
if(NOT IMAGO_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "VERSION must contain a stable major.minor.patch version.")
endif()
project(Imago VERSION ${IMAGO_VERSION} LANGUAGES CXX RC)
configure_file(src/version.h.in generated/version.h @ONLY)
configure_file(resources/app.manifest.in generated/app.manifest @ONLY)

if(NOT WIN32)
  message(FATAL_ERROR "Imago currently targets Windows.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(imago_core STATIC src/image.cpp)
target_include_directories(imago_core PUBLIC src)
target_compile_definitions(imago_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(imago_core PUBLIC user32 gdi32)

add_library(imago_updates STATIC src/updates.cpp src/updates.h src/update_version.h)
target_include_directories(imago_updates PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(imago_updates PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(imago_updates PUBLIC winhttp windowsapp shell32 ole32)

add_executable(Imago WIN32 src/main.cpp src/capture_ui.inl src/capture_extras.inl src/capture_helpers.cpp src/capture_helpers.h src/settings.cpp src/settings.h src/toolbar.cpp src/toolbar.h src/annotations.cpp src/annotations.h src/png.cpp src/png.h resources/app.rc)
target_sources(Imago PRIVATE src/ocr.cpp src/ocr.h src/ocr_ui.inl)
target_sources(Imago PRIVATE src/scroll_capture.cpp src/scroll_capture.h src/scroll_controller.cpp src/scroll_controller.h src/scroll_ui.inl)
target_sources(Imago PRIVATE src/appearance.cpp src/appearance.h src/settings_ui.inl)
target_sources(Imago PRIVATE src/file_io.cpp src/file_io.h src/native_resources.h)
target_sources(Imago PRIVATE src/recording.cpp src/recording.h src/recording_common.h
  src/recording_audio.cpp src/recording_audio.h src/recording_video.cpp src/recording_video.h
  src/recording_writer.cpp src/recording_writer.h src/recording_panel.cpp src/recording_panel.h src/recording_ui.inl)
target_sources(Imago PRIVATE src/recording_devices.cpp src/recording_devices.h)
target_sources(Imago PRIVATE src/recording_border.cpp src/recording_border.h src/recording_files.cpp src/recording_files.h)
target_sources(Imago PRIVATE src/updates_ui.inl)
target_link_libraries(Imago PRIVATE imago_updates)
target_link_libraries(Imago PRIVATE imago_core comctl32 comdlg32 shell32 ole32 dwmapi d2d1 dwrite advapi32 windowscodecs uuid windowsapp)
target_link_libraries(Imago PRIVATE d3d11 dxgi mfplat mfreadwrite mfuuid)
target_include_directories(Imago PRIVATE resources)

include(CTest)
if(BUILD_TESTING)
  add_executable(imago_tests tests/core_tests.cpp)
  target_link_libraries(imago_tests PRIVATE imago_core)
  add_test(NAME imago_core COMMAND imago_tests)
  add_executable(imago_update_tests tests/update_tests.cpp)
  target_link_libraries(imago_update_tests PRIVATE imago_updates)
  add_test(NAME imago_updates COMMAND imago_update_tests)
  add_executable(imago_capture_fixture WIN32 tests/capture_fixture.cpp)
  target_link_libraries(imago_capture_fixture PRIVATE imago_core)
endif()

foreach(target imago_core imago_updates Imago imago_tests imago_update_tests imago_capture_fixture)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- /EHsc)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endif()
endforeach()
if(MSVC)
  target_link_options(Imago PRIVATE /MANIFEST:NO /DELAYLOAD:mfplat.dll /DELAYLOAD:mfreadwrite.dll)
  target_link_libraries(Imago PRIVATE delayimp)
endif()
install(TARGETS Imago RUNTIME DESTINATION .)
